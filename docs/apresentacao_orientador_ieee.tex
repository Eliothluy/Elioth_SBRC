\begin{frame}{O Problema --- ``efici\^encia'' que vem de parar de entregar tr\'afego}
  \begin{beamercolorbox}[wd=\textwidth,sep=3pt]{block body}
    \centering\small
    \textbf{Analogia:} fechar a loja economiza a conta de luz --- mas isso n\~ao \'e
    efici\^encia. \textbf{Efici\^encia \'e vender o mesmo com menos luz.} A literatura
    recente de \emph{energy saving} em O-RAN --- incluindo Bordin et al.\ (CCNC
    2025) --- n\~ao separa as duas coisas.
  \end{beamercolorbox}
  \vspace{1.5mm}
  \begin{center}
  \begin{tikzpicture}[
      box/.style={rectangle,rounded corners=3pt,draw=ieeenavy,fill=ieeenavy!7,

    \node[box] (c) at (4.75,-0.2){\textbf{3.} a ``energia'' cai junto\\ (proxy $=$ tr\'afego carregado)};
    \node[box] (d) at (0,-0.2)   {\textbf{4.} o \emph{reward} sobe:\\ a Eq.~(2) paga por desligar};

  aparece apenas na \emph{avalia\c{c}\~ao} --- \textbf{nunca no treino}.}
\end{frame}

  \begin{columns}[T]

      \end{alertblock}
    \end{column}
    \begin{column}{0.32\textwidth}

  \end{beamercolorbox}
\end{frame}

\begin{frame}{Frente 2 --- CMDP: a Ideia em Uma Frase}
  \begin{beamercolorbox}[wd=\textwidth,sep=4pt]{block body example}
    \centering\large
    ``O b\^onus de energia s\'o \'e pago quando o servi\c{c}o est\'a cumprido ---
    como o b\^onus de um vendedor, atrelado \`a nota de satisfa\c{c}\~ao do cliente.''
  \end{beamercolorbox}
  \vspace{3mm}
  \begin{equation*}
    r' \;=\; r \;-\;
    \textcolor{alertred}{\lambda}\cdot
    \underbrace{\max\!\big(0,\ \tau_{\text{SLA}} - \bar{\rho}^{(W)}\big)}_{\text{quanto o SLA est\'a sendo violado na \'ultima janela}}
  \end{equation*}
  \vspace{1mm}
  \begin{itemize}
    \item \textcolor{alertred}{$\lambda$} \'e o ``tamanho da multa'': a grade fixa
      $\lambda \in \{0;\,0{,}5;\,1;\,2;\,5;\,10\}$ gera \textbf{um ponto na
      fronteira de Pareto por valor} --- do sem-restri\c{c}\~ao ao conservador.
    \item \textbf{Via alternativa (porta suave, Eq.~5.1\,$'$):} em vez de multar,
      \emph{desliga o pr\^emio} --- os termos de energia ($w_2$) s\'o entram no
      reward quando o SLA passa.
  \end{itemize}
  \vspace{1mm}
  {\small Ambas implementadas e testadas: \textbf{87 testes aprovados}; detalhes
  e a auditoria adversarial no pr\'oximo slide.}
\end{frame}
% [SPEAKER NOTE]: "Se o orientador levar UMA frase deste slide, que seja esta:
% o bônus de energia só é pago com o serviço cumprido. A equação é uma multa
% controlada por lambda — e varremos uma grade de lambdas, cada um virando um
% ponto na fronteira, do agressivo ao conservador. A via alternativa nem
% multa: simplesmente desliga o prêmio quando o SLA falha. Os detalhes
% técnicos — por que o dual clássico diverge — vêm no próximo slide."

\begin{frame}{Frente 2 (detalhes) --- Grade Lagrangiana e Porta Suave Auditadas (Rev.\ 1)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Por que o dual primal--dual ingênuo diverge \emph{offline}}
        \small Custo hinge médio do dataset: $\bar{c}_{\mathcal{D}} = 0{,}0757
        > \delta = 0{,}05$ $\Rightarrow$ deriva dual $+2{,}57\times10^{-4}$/passo;
        $\lambda \approx 12{,}9$ em 50k passos e \textbf{satura
        $\lambda_{\max}{=}100$ em ${\approx}388$k} (dentro do orçamento de
        1{,}2M). Em $\lambda{=}12{,}9$: penalidade $\lambda c$ até
        $\mathbf{9{,}1}$ $\gg |r| \approx 0{,}1$--$0{,}5$. Com dataset fixo, a
        média de lote é a constante do comportamento --- o dual não persegue
        ponto de sela.
      \end{alertblock}
      \begin{block}{Ajustes blindados A1--A5 (parecer adversarial incorporado)}
        \small A1: regra de coerência $\delta \times \bar{c}_{\mathcal{D}}$
        pré-treino; A2: crítico de custo $V_c(s)$ + $\hat{J}_c$ sob a política
        corrente (FQE/MC, Robbins--Monro com teto); A5: correção do wrapper
        PPO-SB3.
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{exampleblock}{Via 1 --- Grade Lagrangiana \emph{offline} (A3--A4)}
        \small
        \begin{equation*}
          r_\lambda = r - \lambda\,\max\!\big(0,\ \tau_{\text{SLA}} -
          \bar{\rho}^{(W)}\big)
        \end{equation*}
        Grade fixa $\lambda \in \{0;\,0{,}5;\,1;\,2;\,5;\,10\}$,
        $\lambda_{\max} = 10$, guarda $\lambda\bar{c}/|\bar{r}| \le 5$.
        \textbf{Cada $\lambda$ = um ponto candidato na fronteira de Pareto}
        (treino offline é barato; caro é o live).
      \end{exampleblock}
      \begin{exampleblock}{Via 2 --- Porta suave de SLA, Eq.~(5.1\,$'$)}
        \small
        \begin{equation*}
          r' = w_1\rho - w_3\varsigma - w_4\delta
               - g_{\tau}\!\big(\bar{\rho}^{(W)}\big)\cdot
                 \big(\textcolor{alertred}{w_2\hat{\gamma}} +
                 \textcolor{alertred}{w_2\hat{B}_{sON}}\big)
        \end{equation*}
        \textbf{Ambos} os canais de $w_2$ entram \emph{dentro} do gate: fora
        dele, a política deixa de ser paga por desligar BS violando o SLA.
        Porta suave = padrão (porta dura esparsa em 25--34\% das indicações,
        $W{=}10$). Estado aumentado com a janela $\bar{\rho}^{(W)}$
        (markovianidade): \texttt{full+gate} 86 / \texttt{compact+gate} 79 /
        \texttt{compact70+gate} 71 dims. \ \emph{Suíte: 87 testes aprovados.}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
% [SPEAKER NOTE]: "Submetemos o próprio design a um parecer adversarial e ele
% derrubou o dual clássico: com dataset fixo o multiplicador satura e a
% penalidade engole o reward — números medidos, não opinião. A resposta está à
% direita e já está codificada com 87 testes verdes: uma grade de lambdas (cada
% um vira um ponto no Pareto) e a porta de SLA que coloca os DOIS termos de w2
% dentro do gate. É correção estrutural, não um band-aid de pesos."
